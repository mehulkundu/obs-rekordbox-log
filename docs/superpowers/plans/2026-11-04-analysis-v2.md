**Goal:** Turn the raw controller log (7-20 MB an hour) into `edit.md`, a sheet of at most 300 lines for an hour that editors and AI agents can use with the video. It holds lower thirds, labelled transitions with evidence, edit cues, energy, checks, every clip (ranked and tickable) and YouTube chapters, plus `clips.json`, `moves.jsonl` and a `window` command for detail.

**Architecture:** The v1 analysis (`analysis/rekordbox_log.py`) keeps finding tracks, history names and transition windows, but now in video time from the start. Four new modules sit on top:
- `moves.py` collapses events into gestures and gives any control's value at any time.
- `techniques.py` holds a priority-ordered table of rules (bass swap, drop swap, wordplay, ...), plus moments across the set.
- `audio.py` finds drops, breakdowns, silence and energy in the video's audio with ffmpeg and numpy.
- `sheet.py` builds the clips and `edit.md`.

The plugin and the raw log don't change.

**Tech Stack:** Python 3.10+ standard library. Optional: numpy and ffmpeg for audio, pyrekordbox for names.

```python
def fade(ctl, t0, t1, a, b, step=0.05):
    n = max(1, int(round((t1 - t0) / step)))
    return [c(t0 + i * (t1 - t0) / n, ctl, round(a + (b - a) * i / n, 3)) for i in range(n + 1)]
```

```python
"""Moves: the controller log collapsed to one line per gesture, and a timeline of control values."""
from bisect import bisect_right

GAP = 0.5  # seconds between events that still count as one move
RELATIVE = ('jog', 'jog.side', 'jog.search', 'browse')  # these send signed steps, not positions
POSITION = ('fader', 'crossfader', 'eq.hi', 'eq.mid', 'eq.low', 'trim', 'cfx', 'tempo', 'level',
            'headphone.mix', 'master', 'fx.beat.channel')  # the FX channel lever: its value is the deck
DEFAULTS = {'fader': 1.0}  # every other position control starts centred (0.5)


def kind(ctl):
    if ctl.endswith(RELATIVE):
        return 'relative'
    if ctl.endswith(POSITION):
        return 'position'
    return 'button'


def find_moves(ctls):
    """One move per gesture: a control moving one way with gaps under GAP, a run of jog steps one
    way, or a button press. ctls must be in time order."""
    moves, last = [], {}
    for e in ctls:
        c, t, v = e['ctl'], round(e['t'], 3), e['v']
        k = kind(c)
        if k == 'button':
            if v == 1:
                moves.append({'start': t, 'end': t, 'ctl': c, 'n': 1})
            continue
        m = last.get(c)
        recent = m is not None and t - m['end'] < GAP
        if k == 'relative':
            if recent and (m['steps'] > 0) == (v > 0):
                m['end'], m['steps'], m['n'] = t, m['steps'] + int(v), m['n'] + 1
            else:
                m = last[c] = {'start': t, 'end': t, 'ctl': c, 'steps': int(v), 'n': 1}
                moves.append(m)
            continue
        v = round(v, 3)
        if recent:
            step = v - m['to']
            if step == 0 or m['_dir'] == 0 or (step > 0) == (m['_dir'] > 0):
                m['end'], m['to'], m['n'] = t, v, m['n'] + 1
                m['_dir'] = m['_dir'] or step
                continue
        frm = m['to'] if m else v
        m = last[c] = {'start': t, 'end': t, 'ctl': c, 'from': frm, 'to': v, 'n': 1, '_dir': v - frm}
        moves.append(m)
    for m in moves:
        m.pop('_dir', None)
    return moves


def xfader_gain(deck, x):
    # Both decks full at the centre, cut at the far end, like Rekordbox's default curve.
    return min(1.0, 2 * (1 - x)) if deck == 1 else min(1.0, 2 * x)


class Timeline:
    """Any control's value, a deck's playing state, and a deck's level at any time t.
    MIDI only reports changes, so untouched controls are at their usual spot: channel faders up,
    crossfader and knobs centred, decks stopped."""

    def __init__(self, ctls):
        self.values = {}  # ctl -> ([t], [v])
        self.playing = {1: ([], []), 2: ([], [])}
        self.fx_on = ([], [])
        state, fx = {1: False, 2: False}, False
        for e in ctls:
            c, t, v = e['ctl'], e['t'], e['v']
            if kind(c) == 'position':
                ts, vs = self.values.setdefault(c, ([], []))
                ts.append(t)
                vs.append(v)
            if v != 1:
                continue
            for d in (1, 2):
                if c == f'deck{d}.play':
                    state[d] = not state[d]
                elif c in (f'deck{d}.cue', f'deck{d}.load'):
                    state[d] = False
                else:
                    continue
                self.playing[d][0].append(t)
                self.playing[d][1].append(state[d])
            if c == 'fx.beat.on':  # each press toggles
                fx = not fx
                self.fx_on[0].append(t)
                self.fx_on[1].append(fx)

    @staticmethod
    def _at(series, t, default):
        ts, vs = series
        i = bisect_right(ts, t)
        return vs[i - 1] if i else default

    def value(self, ctl, t):
        default = DEFAULTS.get(ctl.rsplit('.', 1)[-1], 0.5)
        return self._at(self.values.get(ctl, ([], [])), t, default)

    def is_playing(self, d, t):
        return self._at(self.playing[d], t, False)

    def fx_is_on(self, t):
        return self._at(self.fx_on, t, False)

    def level(self, d, t):
        if not self.is_playing(d, t):
            return 0.0
        return self.value(f'deck{d}.fader', t) * xfader_gain(d, self.value('mixer.crossfader', t))

    def samples(self, t0, t1, step=0.1):
        n = max(0, int(round((t1 - t0) / step)) + 1)
        return [round(t0 + i * step, 3) for i in range(n)]


def stamp(seconds, tenths=False):
    """m:ss or h:mm:ss, with a decimal when tenths is set."""
    if seconds is None:
        return '?'
    s = max(0.0, seconds)
    h, rest = divmod(int(s), 3600)
    m, sec = divmod(rest, 60)
    frac = f'.{int((s - int(s)) * 10)}' if tenths else ''
    return f'{h}:{m:02}:{sec:02}{frac}' if h else f'{m}:{sec:02}{frac}'


def parse_time(text):
    """'14:02', '1:02:03' or '842.5' to seconds."""
    parts = [float(p) for p in text.split(':')]
    total = 0.0
    for p in parts:
        total = total * 60 + p
    return total


def window_lines(moves, t0, t1):
    out = []
    for m in moves:
        if m['end'] < t0 or m['start'] > t1:
            continue
        span = stamp(m['start'], True) if m['start'] == m['end'] else f"{stamp(m['start'], True)}-{stamp(m['end'], True)}"
        if 'steps' in m:
            what = f"{m['steps']:+d} steps"
        elif 'from' in m:
            what = f"{m['from']:.2f} -> {m['to']:.2f}"
        else:
            what = 'press'
        out.append(f"{span}  {m['ctl']}  {what}  ({m['n']})")
    return out
```

```python
"""Transition styles and set moments, recognised from moves and deck levels.

Each rule takes a Window and returns None or a match: {style, moment, confidence, evidence}.
Thresholds are starting points from the spec (docs/superpowers/specs/2026-11-04-analysis-v2-design.md);
test_analysis.py pins each one with a case that just meets it and one that just misses.
"""
from moves import stamp

AUDIBLE = 0.1
OVERLAP = 0.3
LOOP_CTLS = ('loop.in', 'loop.half')


class Window:
    """One transition: A (outgoing) and B (incoming) decks, its times, moves and levels."""

    def __init__(self, tl, moves, A, B, start, swap, end, next_start=None, drops=()):
        self.tl, self.A, self.B = tl, A, B
        self.start, self.swap, self.end = start, swap, end
        anchor = start if start is not None else swap
        self.t0 = anchor - 10
        if end is not None:
            self.t1 = end + 5
        else:
            stops = [t for t in ((swap + 20) if swap is not None else None, next_start) if t is not None]
            self.t1 = min(stops) if stops else anchor + 30
        self.moves = [m for m in moves if self.t0 <= m['start'] <= self.t1]
        self.times = tl.samples(self.t0, self.t1)
        self.la = [tl.level(A, t) for t in self.times]
        self.lb = [tl.level(B, t) for t in self.times]
        self.drops = list(drops)

    def deck_moves(self, d, suffix):
        return [m for m in self.moves if m['ctl'] == f'deck{d}.{suffix}']

    def presses(self, d, test):
        return [m['start'] for m in self.moves if m['ctl'].startswith(f'deck{d}.') and test(m['ctl'][6:])]

    def overlap(self):
        return round(sum(0.1 for a, b in zip(self.la, self.lb) if a > OVERLAP and b > OVERLAP), 1)

    def overlap_span(self):
        both = [t for t, a, b in zip(self.times, self.la, self.lb) if a > OVERLAP and b > OVERLAP]
        return (both[0], both[-1]) if both else (None, None)

    def a_out_after(self, t, within):
        """First time in (t, t + within] when A's level is below AUDIBLE, or None."""
        return next((x for x, a in zip(self.times, self.la) if t < x <= t + within and a < AUDIBLE), None)


def match(style, moment, evidence, confidence='medium'):
    return {'style': style, 'moment': round(moment, 3), 'confidence': confidence, 'evidence': evidence}


def is_loop(c):
    return c in LOOP_CTLS or c.startswith('pad.beatloop.')


def is_rolling(c):
    return c == 'loop.half' or c.startswith('pad.beatloop.')


def runs(times, count, within):
    """Runs of at least `count` times inside `within` seconds: [(first, last, n)], not overlapping."""
    out, i = [], 0
    times = sorted(times)
    while i + count - 1 < len(times):
        if times[i + count - 1] - times[i] <= within:
            j = i + count - 1
            while j + 1 < len(times) and times[j + 1] - times[i] <= within:
                j += 1
            out.append((times[i], times[j], j - i + 1))
            i = j + 1
        else:
            i += 1
    return out


# --- transitions, in priority order -------------------------------------------------------------

def wordplay(w):
    if w.start is None:
        return None
    before = [t for t in w.presses(w.A, is_loop) if w.start - 8 <= t <= w.start]
    cues = {}
    for m in w.moves:
        if m['ctl'].startswith(f'deck{w.A}.pad.hotcue.') and w.start - 8 <= m['start'] <= w.start:
            cues.setdefault(m['ctl'], []).append(m['start'])
    groups = [before] + list(cues.values())
    best = max(groups, key=len)
    if len(best) >= 3 and w.start - max(best) <= 1:
        return match('wordplay', w.start, f'deck{w.A} {len(best)} loop/cue presses in the 8 s before deck{w.B} '
                     f'came in, the last {w.start - max(best):.1f} s before')
    return None


def loop_roll(w):
    for first, last, n in runs(w.presses(w.A, is_rolling), 3, 8):
        out = w.a_out_after(last, 4)
        if out is not None:
            return match('loop roll', last, f'deck{w.A} {n} loop presses in {last - first:.1f} s, '
                         f'then out {out - last:.1f} s later')
    return None


def spinback(w):
    jogs = [m for m in w.deck_moves(w.A, 'jog') if m['steps'] < 0]
    for i, m in enumerate(jogs):
        burst = [x for x in jogs[i:] if x['start'] - m['start'] <= 1]
        steps = sum(x['steps'] for x in burst)
        if steps <= -40:
            end = burst[-1]['end']
            if w.a_out_after(end, 1) is not None:
                return match('spinback', m['start'], f'deck{w.A} jog {steps} steps in under 1 s, then out')
    return None


def double_drop(w):
    """Both decks full with both lows up for 4 s, with an audio drop in that stretch. Without the
    audio this is just a blend with the EQs left alone."""
    if not w.drops:
        return None
    run = None
    for t, a, b in zip(w.times, w.la, w.lb):
        full = (a >= 0.8 and b >= 0.8 and w.tl.value(f'deck{w.A}.eq.low', t) >= 0.4
                and w.tl.value(f'deck{w.B}.eq.low', t) >= 0.4)
        if full:
            run = run if run is not None else t
            drop = next((d for d in w.drops if run <= d <= t), None)
            if t - run >= 4 and drop is not None:
                return match('double drop', drop, f'both decks full with both lows up, an audio drop at {stamp(drop)}',
                             'high')
        else:
            run = None
    return None


def drop_swap(w):
    k = 3  # 0.3 s in samples
    falls = [i for i in range(k, len(w.times)) if w.la[i - k] - w.la[i] >= 0.7]
    rises = [i for i in range(k, len(w.times)) if w.lb[i] - w.lb[i - k] >= 0.7]
    for i in falls:
        j = next((j for j in rises if abs(j - i) <= k), None)
        if j is None:
            continue
        cut = w.times[min(i, j)]
        before = sum(0.1 for t, a, b in zip(w.times, w.la, w.lb) if t < cut and a > OVERLAP and b > OVERLAP)
        if before < 2:
            drop = any(abs(d - cut) <= 1 for d in w.drops)
            return match('drop swap', cut, f'deck{w.A} level {w.la[i - k]:.2f}->{w.la[i]:.2f} and deck{w.B} '
                         f'{w.lb[j - k]:.2f}->{w.lb[j]:.2f} within 0.3 s' + (', on an audio drop' if drop else ''),
                         'high' if drop else 'medium')
    return None


def echo_out(w):
    out = w.end
    if out is None:
        return None
    for m in w.moves:
        if m['ctl'] == 'fx.beat.on' and out - 8 <= m['start'] <= out and w.tl.fx_is_on(m['start']) \
                and w.tl.value('fx.beat.channel', m['start']) == w.A:
            return match('echo out', out, f'beat FX on for deck{w.A} {out - m["start"]:.1f} s before it went out')
    return None


def find_scratch(moves, d, t0=None, t1=None):
    jogs = [m['start'] for m in moves if m['ctl'] == f'deck{d}.jog' and (t0 is None or t0 <= m['start'] <= t1)]
    chops = [m['start'] for m in moves if m['ctl'] in (f'deck{d}.fader', 'mixer.crossfader')]
    out = []
    for first, last, n in runs(jogs, 10, 5):
        c = sum(1 for t in chops if first <= t <= first + 5)
        if c >= 4:
            out.append((first, n, c))
    return out


def scratch(w):
    for d in (w.A, w.B):
        for first, n, c in find_scratch(w.moves, d):
            return match('scratch', first, f'deck{d} {n} jog moves and {c} fader/crossfader moves within 5 s')
    return None


def filter_out(w):
    """A's filter goes 0.3 or more off centre while A fades: its level 4 s after is 0.2 lower."""
    for m in w.deck_moves(w.A, 'cfx'):
        if abs(m['to'] - 0.5) >= 0.3 and w.tl.level(w.A, m['end'] + 4) <= w.tl.level(w.A, m['start']) - 0.2:
            return match('filter out', m['end'], f"deck{w.A} filter {m['from']:.2f}->{m['to']:.2f} while it faded out")
    return None


def filter_in(w):
    """B's filter comes back to centre from 0.3 or more off while B rises: 0.2 louder than 4 s before."""
    for m in w.deck_moves(w.B, 'cfx'):
        if abs(m['from'] - 0.5) >= 0.3 and abs(m['to'] - 0.5) <= 0.05 and \
                w.tl.level(w.B, m['end']) >= w.tl.level(w.B, m['start'] - 4) + 0.2:
            return match('filter in', m['end'], f"deck{w.B} filter {m['from']:.2f}->{m['to']:.2f} as it came in")
    return None


SWAP_WITHIN = 4  # seconds between B's knob going up and A's going down: about 2 bars at 128 BPM


def band_swaps(w, band):
    """Times when B's band comes up from cut while A's goes down, within SWAP_WITHIN of each other."""
    ups = [m for m in w.deck_moves(w.B, f'eq.{band}') if m['from'] <= 0.2 and m['to'] > 0.4]
    downs = [m for m in w.deck_moves(w.A, f'eq.{band}') if m['from'] > 0.2 and m['to'] < 0.2]
    out = []
    for u in ups:
        d = next((d for d in downs if abs(d['end'] - u['end']) <= SWAP_WITHIN), None)
        if d:
            out.append((max(u['end'], d['end']), u, d))
    return out


def three_band(w):
    found = {band: band_swaps(w, band) for band in ('low', 'mid', 'hi')}
    if not all(found.values()):
        return None
    times = sorted(found[b][-1][0] for b in found)
    if all(b - a >= 2 for a, b in zip(times, times[1:])):
        low = found['low'][-1][0]
        return match('three-band swap', low, 'lows, mids and highs swapped one at a time: ' +
                     ', '.join(f'{b} {stamp(found[b][-1][0])}' for b in ('low', 'mid', 'hi')))
    return None


def bass_swap(w):
    swaps = band_swaps(w, 'low')
    if not swaps:
        return None
    t, u, d = swaps[-1]
    more = f', {len(swaps)} swaps in all' if len(swaps) > 1 else ''
    return match('bass swap', t, f"deck{w.B} low {u['from']:.2f}->{u['to']:.2f} while deck{w.A} low "
                 f"{d['from']:.2f}->{d['to']:.2f} within {abs(u['end'] - d['end']):.1f} s{more}")


def tempo_ramp(w):
    a, b = w.overlap_span()
    if a is None:
        return None
    for d in (w.A, w.B):  # moves, not values: an untouched slider's real position is unknown
        ramp = [m for m in w.deck_moves(d, 'tempo') if a <= m['start'] <= b]
        if ramp and abs(ramp[-1]['to'] - ramp[0]['from']) >= 0.05:
            return match('tempo ramp', (ramp[0]['start'] + ramp[-1]['end']) / 2,
                         f"deck{d} tempo {ramp[0]['from']:.2f}->{ramp[-1]['to']:.2f} during the overlap")
    return None


def crossfader_cut(w):
    big = [m['start'] for m in w.moves if m['ctl'] == 'mixer.crossfader' and abs(m['to'] - m['from']) > 0.5]
    for first, last, n in runs(big, 4, 3):
        return match('crossfader cut', first, f'{n} crossfader moves of more than half its travel within 3 s')
    return None


def long_blend(w):
    o = w.overlap()
    if o >= 16 and w.swap is not None:
        return match('long blend', w.swap, f'both decks up for {o:.0f} s')
    return None


def quick_mix(w):
    o = w.overlap()
    if o < 4 and w.swap is not None:
        return match('quick mix', w.swap, f'both decks up for {o:.1f} s')
    return None


# Tricks first, then how the decks were mixed, then how the old track left, then blends.
TRANSITIONS = [wordplay, loop_roll, spinback, double_drop, drop_swap, scratch, three_band, bass_swap,
               echo_out, filter_out, filter_in, tempo_ramp, crossfader_cut, long_blend, quick_mix]
BLENDS = ('long blend', 'quick mix')


def label_transition(w):
    found = [m for m in (rule(w) for rule in TRANSITIONS) if m]
    if not found:
        moment = w.swap if w.swap is not None else w.start
        return {'style': 'unusual', 'styles': [], 'moment': moment, 'confidence': 'low',
                'evidence': 'no known pattern: see the moves'}
    top = found[0]
    return {**top, 'styles': [m['style'] for m in found], 'matches': found}


def shown_styles(styles):
    """How a transition reads in the sheet: every style, without the blend kinds when others matched."""
    main = [s for s in styles if s not in BLENDS]
    return ' + '.join(main or styles) or 'unusual'


# --- moments across the set ------------------------------------------------------------------

COVERED = {'loop roll': ('loop roll', 'wordplay'), 'scratch': ('scratch',), 'filter sweep': ('filter out', 'filter in'),
           'beat fx': ('echo out',)}


def find_moments(moves, tl, transitions, end):
    found = []
    for d in (1, 2):
        rolls = [m['start'] for m in moves if m['ctl'].startswith(f'deck{d}.') and is_loop(m['ctl'][6:])]
        for first, last, n in runs(rolls, 3, 8):
            found.append(match('loop roll', last, f'deck{d} {n} loop presses in {last - first:.1f} s') | {'deck': d})
        for first, n, c in find_scratch(moves, d):
            found.append(match('scratch', first, f'deck{d} {n} jog moves and {c} fader/crossfader moves within 5 s') | {'deck': d})
        cues = [m['start'] for m in moves if m['ctl'].startswith(f'deck{d}.pad.hotcue.')]
        for first, last, n in runs(cues, 6, 8):
            found.append(match('hot cue juggling', first, f'deck{d} {n} hot cue presses in {last - first:.1f} s') | {'deck': d})
        hits = [m['start'] for m in moves if m['ctl'].startswith(f'deck{d}.pad.sampler.')]
        for group in groups(hits, 8):
            found.append(match('sampler hit', group[0], f'deck{d} {len(group)} sampler pad presses') | {'deck': d})
        sweeps = [m['start'] for m in moves if m['ctl'] == f'deck{d}.cfx' and abs(m['to'] - 0.5) >= 0.3]
        for group in groups(sweeps, 8):
            found.append(match('filter sweep', group[0], f'deck{d} {len(group)} filter sweeps in '
                               f'{group[-1] - group[0]:.0f} s') | {'deck': d})
    for m in moves:
        if m['ctl'] == 'fx.beat.on' and tl.fx_is_on(m['start']):
            off = next((x['start'] for x in moves if x['ctl'] == 'fx.beat.on' and x['start'] > m['start']), end)
            ch = tl.value('fx.beat.channel', m['start'])
            found.append(match('beat fx', m['start'], f'beat FX on for {off - m["start"]:.1f} s'
                               + (f' on deck{int(ch)}' if ch in (1, 2) else '')))
    # 3 or more drop swaps in a row, each track on for under 60 s
    row = []
    for tr in transitions + [None]:
        quick = tr is not None and tr.get('style') == 'drop swap' and (not row or tr['moment'] - row[-1]['moment'] < 60)
        if quick:
            row.append(tr)
            continue
        if len(row) >= 3:
            found.append(match('drop mixing', row[0]['moment'], f'{len(row)} drop swaps in a row, '
                               f'{stamp(row[0]["moment"])}-{stamp(row[-1]["moment"])}'))
        row = [tr] if tr is not None and tr.get('style') == 'drop swap' else []
    # skip a moment a transition already describes: the same kind of match within 5 s
    keep = []
    for m in found:
        near = [x for t in transitions for x in t.get('matches', []) if abs(x['moment'] - m['moment']) <= 5]
        if any(x['style'] in COVERED.get(m['style'], ()) for x in near):
            continue
        keep.append(m)
    return sorted(keep, key=lambda m: m['moment'])


def groups(times, gap):
    """Split sorted times wherever two are `gap` seconds or more apart."""
    out = []
    for t in sorted(times):
        if out and t - out[-1][-1] < gap:
            out[-1].append(t)
        else:
            out.append([t])
    return out
```

```python
"""Drops, breakdowns, silence and energy from the recording's audio (needs numpy and ffmpeg)."""
import shutil
import subprocess
from pathlib import Path

SR = 11025
HOP = 0.1          # seconds between frames
FRAME = 0.2        # seconds of audio per frame
VIDEO_EXTS = ('.mov', '.mp4', '.mkv', '.flv', '.ts', '.m4v')


def find_video(log_path):
    base = log_path.name.removesuffix('.rekordbox.jsonl')
    for ext in VIDEO_EXTS:
        p = log_path.with_name(base + ext)
        if p.exists():
            return p
    return None


def read_audio(video):
    """Mono float samples at SR, or (None, reason)."""
    if shutil.which('ffmpeg') is None:
        return None, 'ffmpeg not installed'
    import numpy as np
    run = subprocess.run(['ffmpeg', '-v', 'error', '-i', str(video), '-vn', '-ac', '1', '-ar', str(SR),
                          '-f', 'f32le', '-'], capture_output=True)
    if run.returncode != 0:
        return None, f'ffmpeg could not read {video.name}: {run.stderr.decode(errors="replace").strip()[:200]}'
    x = np.frombuffer(run.stdout, dtype='<f4')
    if x.size == 0:
        return None, f'{video.name} has no audio track'
    return x, None


def spans(mask, min_frames):
    """(start, end) frame indices of runs of True at least min_frames long."""
    out, i, n = [], 0, len(mask)
    while i < n:
        if mask[i]:
            j = i
            while j < n and mask[j]:
                j += 1
            if j - i >= min_frames:
                out.append((i, j))
            i = j
        else:
            i += 1
    return out


def analyse_audio(x, sr=SR):
    """Features and events from mono samples. Times are seconds from the start of the audio."""
    import numpy as np
    if np.abs(x).max() < 10 ** (-60 / 20):
        return None, 'the audio is silent (is Rekordbox captured in OBS?)'
    hop, size = int(sr * HOP), int(sr * FRAME)
    n = max(1, 1 + (len(x) - size) // hop)
    pad = np.pad(x, (size // 2, size))
    win = np.hanning(size)
    freqs = np.fft.rfftfreq(size, 1 / sr)
    band = (freqs >= 30) & (freqs <= 150)
    bass, rms = np.empty(n), np.empty(n)
    for i in range(n):  # frame i is centred on i * HOP seconds
        f = pad[i * hop:i * hop + size]
        bass[i] = (np.abs(np.fft.rfft(f * win))[band] ** 2).sum()
        rms[i] = np.sqrt(np.mean(f ** 2))
    # The dips between kicks must not count: a 0.6 s moving max bridges them at 100 BPM and up,
    # then a 1 s moving mean evens it out.
    m = int(round(0.6 / HOP))
    peak = np.lib.stride_tricks.sliding_window_view(np.pad(bass, (m // 2, m - 1 - m // 2), mode='edge'), m).max(axis=1)
    k = int(round(1 / HOP))
    smooth = np.convolve(peak, np.ones(k) / k, mode='same')
    F = 0.7 * np.percentile(smooth, 95)
    full = smooth >= F

    def count(mask, a, b):
        return int(mask[max(0, a):max(0, b)].sum())

    held = np.convolve(full, np.ones(20), mode='full')[19:len(full) + 19] >= 20  # a 2 s run starts here
    drops, last = [], None
    for i in range(1, n):
        if not full[i] or full[i - 1]:
            continue
        if count(full, i, i + 80) < 80:      # stays full for 8 s
            continue
        if held[max(0, i - 200):max(0, i - 20)].any():  # had a 2 s full stretch in the 20 s before
            continue
        t = i * HOP
        if last is not None and t - last < 6:
            continue
        # back to the first kick: first frame in the 2 s before where raw bass passes 0.5F
        lo = max(0, i - 20)
        first = next((j for j in range(lo, i + 1) if bass[j] >= 0.5 * F), i)
        drops.append(round(first * HOP, 1))
        last = t
    after = int(drops[0] / HOP) if drops else n
    quiet = np.zeros(n, bool)
    quiet[after:] = smooth[after:] < 0.3 * F
    breakdowns = [(round(a * HOP, 1), round(b * HOP, 1)) for a, b in spans(quiet, 160)]
    silence = [(round(a * HOP, 1), round(b * HOP, 1)) for a, b in spans(rms < 10 ** (-50 / 20), 10)]
    per_min = int(round(60 / HOP))
    minutes = [float(smooth[i:i + per_min].mean()) for i in range(0, n, per_min)]
    top = max(minutes) or 1.0
    energy = [min(7, int(m / top * 7 + 0.5)) for m in minutes]
    return {'F': float(F), 'smooth': smooth, 'drops': drops, 'breakdowns': breakdowns,
            'silence': silence, 'energy': energy, 'duration': round(len(x) / sr, 2)}, None


def bass_at(audio, t):
    s = audio['smooth']
    return float(s[min(len(s) - 1, max(0, int(round(t / HOP))))])


def load(log_path, video=None):
    """(audio dict, video path, reason it was skipped)."""
    video = Path(video) if video else find_video(log_path)
    if video is None:
        return None, None, 'no video next to the log (pass --video)'
    try:
        import numpy  # noqa: F401
    except ImportError:
        return None, video, 'numpy not installed (pip install numpy)'
    x, why = read_audio(video)
    if x is None:
        return None, video, why
    audio, why = analyse_audio(x)
    return audio, video, why
```

```python
"""Clips and the edit sheet (edit.md): what an editor or an AI needs, in a few hundred lines."""
from moves import stamp, window_lines
from techniques import shown_styles

WEIGHTS = {
    'drop swap': 3, 'double drop': 3, 'wordplay': 3,
    'loop roll': 2, 'scratch': 2, 'hot cue juggling': 2, 'spinback': 2, 'drop mixing': 2,
    'echo out': 1.5, 'unusual': 1.5,
    'filter out': 1, 'filter in': 1, 'bass swap': 1, 'three-band swap': 1, 'tempo ramp': 1,
    'crossfader cut': 1, 'filter sweep': 1, 'beat fx': 1, 'sampler hit': 1,
    'long blend': 0.5, 'quick mix': 0.5, 'drop': 0,
}
CLOSE_UP = ('loop roll', 'scratch', 'hot cue juggling', 'spinback', 'sampler hit', 'beat fx', 'filter sweep',
            'drop mixing')
TOP_CLIPS = 30
UNUSUAL_LINES = 40
GLYPHS = '▁▂▃▄▅▆▇█'


def label(tr):
    if tr is None:
        return '?'
    if tr.get('title'):
        return f"{tr['title']} - {tr['artist']}" if tr.get('artist') else tr['title']
    return f"Deck {tr['deck']}" if tr.get('deck') else '?'


def on_air(tracks, transitions, duration):
    """Each track's on-air span: from the swap into it (or its start) to the swap out of it."""
    into = {t['into']: t['swap'] if t['swap'] is not None else t['start'] for t in transitions}
    starts = [0.0 if i == 0 else (into.get(i) if into.get(i) is not None else tr['start'])
              for i, tr in enumerate(tracks)]
    return [(s, starts[i + 1] if i + 1 < len(starts) else duration) for i, s in enumerate(starts)]


def playing_at(tracks, spans, t):
    for tr, (a, b) in zip(tracks, spans):
        if a <= t < b:
            return tr
    return tracks[-1] if tracks else None


def make_clips(tracks, transitions, moments, audio, duration):
    drops = audio['drops'] if audio else []
    spans = on_air(tracks, transitions, duration)
    used, cands = set(), []
    for t in transitions:
        swap = t['swap'] if t['swap'] is not None else t['moment']
        if swap is None:
            continue
        a, b = tracks[t['into'] - 1], tracks[t['into']]
        slam = [d for d in drops if abs(d - swap) <= 5 and d not in used]
        later = [d for d in drops if 8 <= d - swap <= 30 and d not in used]
        if slam:
            d = slam[0]
            used.add(d)
            cands.append(('slam', d - 20, d + 12, d, t['style'], a, b))
        elif later:
            d = later[0]
            used.add(d)
            cands.append(('transition+drop', swap - 18, d + 10, d, t['style'], a, b))
        else:
            cands.append(('transition', swap - 20, swap + 20, t['moment'] if t['moment'] is not None else swap,
                          t['style'], a, b))
    for d in drops:
        if d not in used:
            tr = playing_at(tracks, spans, d)
            cands.append(('drop', d - 20, d + 12, d, 'drop', tr, tr))
    for m in moments:
        tr = playing_at(tracks, spans, m['moment'])
        cands.append(('moment', m['moment'] - 10, m['moment'] + 15, m['moment'], m['style'], tr, tr))

    clips = []
    for kind, start, end, moment, style, a, b in sorted(cands, key=lambda c: c[1]):
        reasons, score = [], 0.0
        if kind == 'drop' or any(abs(d - moment) <= 1 for d in drops):
            score += 3
            reasons.append('confirmed drop')
        w = WEIGHTS.get(style, 1)
        score += w
        if style != 'drop':
            reasons.append(style)
        if audio:
            from audio import bass_at
            jump = (bass_at(audio, moment + 5) - bass_at(audio, moment - 5)) / audio['F'] if audio['F'] else 0
            jump = max(0.0, min(2.0, jump))
            score += jump
            if jump >= 1:
                reasons.append('big energy jump')
            elif jump >= 0.3:
                reasons.append('energy jump')
        if a and b and a.get('title') and b.get('title'):
            score += 0.5
            reasons.append('songs named')
        clips.append({'kind': kind, 'start': round(max(0.0, start), 1), 'end': round(min(duration, end), 1),
                      'moment': round(moment, 1), 'style': style, 'score': round(score, 1),
                      'reason': ', '.join(reasons), 'from': label(a), 'to': label(b)})
    for i, c in enumerate(clips):
        c['id'] = f'C{i + 1:02}'
    return [{'id': c.pop('id'), **c} for c in clips]


def stopped_while_audible(moves, tl):
    """Play or cue presses that stopped a deck heard at 0.5 or more for the whole 0.5 s before."""
    out = []
    for m in moves:
        for d in (1, 2):
            t = m['start']
            if m['ctl'] in (f'deck{d}.play', f'deck{d}.cue') and tl.is_playing(d, t - 0.001) \
                    and not tl.is_playing(d, t) and all(tl.level(d, x) >= 0.5 for x in tl.samples(t - 0.5, t - 0.1)):
                out.append((t, d))
    return out
```

```python
#!/usr/bin/env python3
"""Turn an obs-rekordbox-log .jsonl into an edit sheet: tracks, labelled transitions, drops and clips.

    python3 rekordbox_log.py SET.rekordbox.jsonl [--out DIR] [--video FILE]
    python3 rekordbox_log.py window SET.rekordbox.jsonl FROM TO
    python3 rekordbox_log.py --selftest

Writes <name>.analysis/edit.md (for editors and AI), clips.json, moves.jsonl, tracklist.json,
transitions.json, fx.json and chapters.txt. Track names come from Rekordbox's play history through
pyrekordbox; drops come from the video's audio through numpy and ffmpeg. Without either, the rest works.
"""
import argparse
import json
import sys
from datetime import datetime, timedelta
from pathlib import Path


AUDIBLE = 0.1
# Rekordbox adds a track to its history once it has played this long (60.007 s measured on
# Rekordbox 7.2 with a DDJ-FLX4).
HISTORY_DELAY = 60


def load(path):
    header, events = None, []
    with open(path, encoding='utf-8') as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            try:
                e = json.loads(line)
            except json.JSONDecodeError:
                continue  # a crash can leave half a line at the end
            if e.get('type') == 'session':
                header = e
            else:
                events.append(e)
    if header is None:
        sys.exit(f'{path}: no session header, not an obs-rekordbox-log file')
    return header, events


def video_clock(header, events):
    """Returns vt(t): seconds into the video. Adds offset_ms and removes time spent paused."""
    offset = header.get('offset_ms', 0) / 1000
    pauses, paused_at = [], None
    for e in events:
        if e['type'] == 'pause':
            paused_at = e['t']
        elif e['type'] == 'resume' and paused_at is not None:
            pauses.append((paused_at, e['t']))
            paused_at = None
    if paused_at is not None:
        pauses.append((paused_at, float('inf')))

    def vt(t):
        if t is None:
            return None
        gone = sum(min(t, b) - a for a, b in pauses if t > a)
        return round(t + offset - gone, 3)
    return vt


def read_history(header, end_t):
    """(Rekordbox history rows played during the session with t in log seconds, why names are missing)."""
    try:
        from pyrekordbox import MasterDatabase
    except ImportError:
        return [], 'pyrekordbox not installed (pip install pyrekordbox): tracks are named by deck'

    # Rekordbox stores created_at as naive local time
    start = datetime.fromisoformat(header['wall']).astimezone().replace(tzinfo=None)
    end = start + timedelta(seconds=end_t)
    try:
        rows = [s for s in MasterDatabase().get_history_songs().all() if start <= s.created_at <= end]
    except Exception as err:  # database locked, key missing, schema changed
        return [], f'cannot read Rekordbox history ({err}): tracks are named by deck'
    out = []
    for s in sorted(rows, key=lambda s: s.created_at):
        c = s.Content
        if c is None:
            continue
        out.append({
            'title': c.Title, 'artist': c.ArtistName, 'bpm': c.BPM / 100 if c.BPM else None,
            'key': c.KeyName, 'path': c.FolderPath, 't': (s.created_at - start).total_seconds(),
        })
    return out, None


def xfader_gain(deck, x):
    # Both decks full at the centre, cut at the far end, like Rekordbox's default curve.
    return min(1.0, 2 * (1 - x)) if deck == 1 else min(1.0, 2 * x)


class Mixer:
    """Replays controller events. MIDI only reports changes, so untouched controls are assumed
    at their usual spot: channel faders up, crossfader and knobs centred, decks stopped."""

    def __init__(self):
        self.ctl = {}
        self.playing = {1: False, 2: False}

    def apply(self, e):
        name, v = e['ctl'], e['v']
        self.ctl[name] = v
        for d in (1, 2):
            if name == f'deck{d}.play' and v == 1:
                self.playing[d] = not self.playing[d]
            elif name in (f'deck{d}.cue', f'deck{d}.load') and v == 1:
                self.playing[d] = False

    def level(self, d):
        if not self.playing[d]:
            return 0.0
        return self.ctl.get(f'deck{d}.fader', 1.0) * xfader_gain(d, self.ctl.get('mixer.crossfader', 0.5))


def find_tracks(ctls, history, end_t):
    """Every load that was then played is a track. History rows give the names."""
    # By position in the log, not time: a load and a play can land in the same millisecond.
    # `i` is the load's index in ctls; analyse() drops it from the output.
    tracks = []
    for i, e in enumerate(ctls):
        for d in (1, 2):
            if e['ctl'] == f'deck{d}.load' and e['v'] == 1:
                tracks.append({'deck': d, 'load': e['t'], 'start': None, 'i': i})
    for tr in tracks:
        d = tr['deck']
        until = min([x['i'] for x in tracks if x['deck'] == d and x['i'] > tr['i']], default=len(ctls))
        tr['start'] = next((e['t'] for e in ctls[tr['i'] + 1:until]
                            if e['ctl'] == f'deck{d}.play' and e['v'] == 1), None)
    tracks = [tr for tr in tracks if tr['start'] is not None]
    # A deck played before any load in the log was loaded before Record (or with the mouse).
    for d in (1, 2):
        first_load = next((i for i, e in enumerate(ctls) if e['ctl'] == f'deck{d}.load' and e['v'] == 1), len(ctls))
        play = next((i for i, e in enumerate(ctls[:first_load]) if e['ctl'] == f'deck{d}.play' and e['v'] == 1), None)
        if play is not None:
            tracks.append({'deck': d, 'load': None, 'start': ctls[play]['t'], 'i': play})

    used = set()
    for h in history:
        names = {k: h[k] for k in ('title', 'artist', 'bpm', 'key', 'path')}
        free = [i for i, tr in enumerate(tracks) if i not in used and tr['i'] is not None and tr['start'] <= h['t']]
        if free:
            i = min(free, key=lambda i: abs(h['t'] - tracks[i]['start'] - HISTORY_DELAY))
            used.add(i)
            tracks[i].update(names)
        else:  # no load or play press to pair with
            tracks.append({'deck': None, 'load': None, 'start': h['t'], 'i': None, **names})
    # A track Rekordbox hasn't put in its history (it waits about a minute) is kept if it was
    # heard. One never heard was only previewed in headphones.
    tracks = [tr for tr in tracks if tr.get('title') or heard(tr, tracks, ctls)]
    return sorted(tracks, key=lambda tr: tr['start'])


def heard(tr, tracks, ctls):
    """True if the track's deck was audible at some point while the track was on it."""
    if tr['i'] is None:
        return True
    until = min([x['i'] for x in tracks if x['deck'] == tr['deck'] and x['i'] is not None and x['i'] > tr['i']],
                default=len(ctls))
    m = Mixer()
    for j, e in enumerate(ctls[:until]):
        m.apply(e)
        if j >= tr['i'] and m.level(tr['deck']) > AUDIBLE:
            return True
    return False


def label(tr):
    if tr.get('title'):
        return f"{tr['title']} - {tr['artist']}" if tr.get('artist') else tr['title']
    return f"Deck {tr['deck']}"


def find_transitions(tracks, ctls):
    out = []
    for i, (a, b) in enumerate(zip(tracks, tracks[1:])):
        A, B = a['deck'], b['deck']
        if not A or not B or A == B:
            continue
        m = Mixer()
        start = swap = end = None
        for j, e in enumerate(ctls):  # replay from the top so every control has its latest value
            m.apply(e)
            if j < b['i']:
                continue
            la, lb = m.level(A), m.level(B)
            if start is None and lb > AUDIBLE:
                start = e['t']
            if start is not None and swap is None and lb >= la:
                swap = e['t']
            if start is not None and end is None and la < AUDIBLE:
                end = e['t']
            if swap is not None and end is not None:
                break
        out.append({'into': i + 1, 'from': label(a), 'to': label(b), 'from_deck': A, 'to_deck': B,
                    'start': start, 'swap': swap, 'end': end})
    return out


def find_fx(ctls, end_t):
    spans, on, channel, level = [], None, None, None
    for e in ctls:
        n = e['ctl']
        if n == 'fx.beat.channel':
            channel = e['v']
        elif n == 'fx.beat.level':
            level = e['v']
        elif n == 'fx.beat.on' and e['v'] == 1:  # each press toggles
            if on is None:
                on = {'fx': 'beat', 'start': e['t'], 'end': end_t, 'channel': channel, 'level': level}
                spans.append(on)
            else:
                on['end'] = e['t']
                on = None
    for d in (1, 2):
        span = None
        for e in ctls:
            if e['ctl'] != f'deck{d}.cfx':
                continue
            away = abs(e['v'] - 0.5) > 0.05
            if away and span is None:
                span = {'fx': 'cfx', 'start': e['t'], 'end': end_t, 'deck': d, 'peak': e['v']}
                spans.append(span)
            elif away:
                span['peak'] = max(span['peak'], e['v'], key=lambda v: abs(v - 0.5))
            elif span is not None:
                span['end'] = e['t']
                span = None
    return sorted(spans, key=lambda s: s['start'])


def stamp(seconds):
    s = int(seconds)
    h, m, s = s // 3600, s % 3600 // 60, s % 60
    return f'{h}:{m:02}:{s:02}' if h else f'{m}:{s:02}'


def chapters(tracks, transitions):
    into = {t['into']: (t['swap'] if t['swap'] is not None else t['start']) for t in transitions}
    lines, last = [], None
    for i, tr in enumerate(tracks):
        at = 0 if i == 0 else into.get(i, tr['start'])
        if at is None or (last is not None and at - last < 10):  # YouTube: chapters at least 10s long
            continue
        lines.append(f'{stamp(at)} {label(tr)}')
        last = at
    if len(lines) < 3:
        lines.insert(0, f'# YouTube needs at least 3 chapters; this set has {len(lines)}.')
    return '\n'.join(lines) + '\n'


def analyse(header, events, history):
    """history: [{'title', 'artist', 'bpm', 'key', 'path', 't'}] with t in log seconds.
    Everything comes back in video time."""
    vt = video_clock(header, events)
    ctls = [{**e, 't': vt(e['t'])} for e in events if e['type'] == 'ctl']
    end_t = vt(events[-1]['t']) if events else 0
    history = [{**h, 't': vt(h['t'])} for h in history]

    tracks = find_tracks(ctls, history, end_t)
    transitions = find_transitions(tracks, ctls)
    fx = find_fx(ctls, end_t)
    for tr in tracks:
        del tr['i']
    tracklist = [{'n': i + 1, **tr} for i, tr in enumerate(tracks)]
    return {'tracklist': tracklist, 'transitions': transitions, 'fx': fx, 'chapters': chapters(tracks, transitions),
            'ctls': ctls, 'end': end_t}


def build(result, log_name, audio=None, audio_reason=None, video=None, names_reason=None):
    """Label transitions, find moments and clips, and write the edit sheet. Returns
    (moves, moments, clips, edit.md text); result['transitions'] gains style, styles, moment, evidence."""
    from moves import find_moves, Timeline
    from techniques import Window, label_transition, find_moments
    from sheet import make_clips, edit_md

    ctls, end = result['ctls'], result['end']
    moves, tl = find_moves(ctls), Timeline(ctls)
    tracks, transitions = result['tracklist'], result['transitions']
    duration = max(end, audio['duration'] if audio else 0)
    drops = audio['drops'] if audio else []
    for t in transitions:
        if t['start'] is None and t['swap'] is None:
            anchor = tracks[t['into']]['start']
            t.update(style='unusual', styles=[], moment=anchor, confidence='low', t0=anchor - 10, t1=anchor + 30,
                     evidence='the incoming deck never became audible')
            continue
        nxt = tracks[t['into'] + 1]['start'] if t['into'] + 1 < len(tracks) else None
        w = Window(tl, moves, t['from_deck'], t['to_deck'], t['start'], t['swap'], t['end'], nxt, drops)
        t.update(label_transition(w), t0=round(w.t0, 3), t1=round(w.t1, 3))
    moments = find_moments(moves, tl, transitions, end)
    clips = make_clips(tracks, transitions, moments, audio, duration)
    unnamed = sum(1 for tr in tracks if not tr.get('title'))
    names = names_reason or (f'{unnamed} unnamed (Rekordbox adds a track to its history after 60 s of play)'
                             if unnamed else None)
    md = edit_md(log_name, video, duration, tracks, transitions, moments, audio, audio_reason, moves, tl, clips,
                 result['chapters'], names)
    return moves, moments, clips, md


def window(argv):
    """Print the moves between two times: the detail behind any line of the edit sheet."""
    from moves import find_moves, parse_time, window_lines
    if len(argv) != 3:
        sys.exit('usage: rekordbox_log.py window SET.rekordbox.jsonl FROM TO   (m:ss, h:mm:ss or seconds)')
    header, events = load(Path(argv[0]))
    vt = video_clock(header, events)
    ctls = [{**e, 't': vt(e['t'])} for e in events if e['type'] == 'ctl']
    print('\n'.join(window_lines(find_moves(ctls), parse_time(argv[1]), parse_time(argv[2]))) or '(no moves)')


def main():
    if sys.argv[1:2] == ['window']:
        return window(sys.argv[2:])
    import audio as audio_mod
    ap = argparse.ArgumentParser(description=__doc__.splitlines()[0])
    ap.add_argument('log', type=Path)
    ap.add_argument('--out', type=Path, help='output folder (default: next to the log, <name>.analysis)')
    ap.add_argument('--video', type=Path, help='the recording (default: the video next to the log)')
    args = ap.parse_args()

    header, events = load(args.log)
    end_t = events[-1]['t'] if events else 0
    history, names_reason = read_history(header, end_t)
    result = analyse(header, events, history)
    audio, video, audio_reason = audio_mod.load(args.log, args.video)
    moves, moments, clips, md = build(result, args.log.name, audio, audio_reason, video, names_reason)

    out = args.out or args.log.with_name(args.log.name.removesuffix('.rekordbox.jsonl') + '.analysis')
    out.mkdir(parents=True, exist_ok=True)
    for name in ('tracklist', 'transitions', 'fx'):
        (out / f'{name}.json').write_text(json.dumps(result[name], indent=2, ensure_ascii=False) + '\n', encoding='utf-8')
    (out / 'chapters.txt').write_text(result['chapters'], encoding='utf-8')
    (out / 'clips.json').write_text(json.dumps(clips, indent=2, ensure_ascii=False) + '\n', encoding='utf-8')
    (out / 'moves.jsonl').write_text(''.join(json.dumps(m) + '\n' for m in moves), encoding='utf-8')
    (out / 'edit.md').write_text(md, encoding='utf-8')
    print(f"{len(result['tracklist'])} tracks, {len(result['transitions'])} transitions, {len(moments)} moments, "
          f"{len(clips)} clips, audio: {'yes' if audio else audio_reason} -> {out / 'edit.md'}")


def selftest():
    header = {'type': 'session', 'v': 1, 'wall': '2026-11-03T21:00:00.000+05:30', 'offset_ms': 0}

    assert len(r['transitions']) == 1

    # A deck loaded before Record starts a track at its first play. A track never heard (fader

    ev = [c(1, 'deck1.play', 1), c(5, 'deck2.load', 1), c(6, 'deck2.fader', 0.0), c(7, 'deck2.play', 1),
```
